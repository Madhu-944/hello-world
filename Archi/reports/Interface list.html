Overview of application interfaces

select I.name AS Interface
     , I.documentation AS Description
     , P.name AS Platform
     , BO.name AS BusinessObject
     , CASE 
        WHEN RIA.targetid = I.id THEN 'IN'
        WHEN RIA.sourceid = I.id THEN 'OUT'
       END AS FlowDirection
    , CASE 
        WHEN RIA.targetid = I.id THEN IN_APP.name
        WHEN RIA.sourceid = I.id THEN OUT_APP.name
      END AS Application
FROM Elements I
LEFT JOIN Relationships RP 
    ON RP.targetid = I.id 
   AND RP.type = 'RealizationRelationship'
LEFT JOIN Elements P 
    ON RP.sourceid = P.id
LEFT JOIN Relationships RBO 
    ON (RBO.sourceid = I.id OR RBO.targetid = I.id)
   AND RBO.type = 'AssociationRelationship'
LEFT JOIN Elements BO 
    ON (BO.id = RBO.sourceid OR BO.id = RBO.targetid)
   AND BO.id <> I.id
LEFT JOIN Relationships RIA 
    ON (RIA.sourceid = I.id OR RIA.targetid = I.id)
   AND RIA.type = 'FlowRelationship'
LEFT JOIN Elements IN_APP 
    ON RIA.sourceid = IN_APP.id
LEFT JOIN Elements OUT_APP 
    ON RIA.targetid = OUT_APP.id
WHERE I.type = 'ApplicationInterface'
order by 1,4
