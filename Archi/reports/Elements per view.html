Checklist elements per view

SELECT
  v.name AS View,
  COUNT(vc.contentid) AS Element_count
FROM Views v
JOIN ViewsContent vc ON v.id = vc.viewid
JOIN Elements e ON vc.contentid = e.id
WHERE 1=1
GROUP BY v.name, v.id
ORDER BY elementCount DESC
